/* =========================================================
   BLOOM MUSIC ENGINE
   ========================================================= */

const bloomMusic =
    document.getElementById("bloomMusic");

let musicFadeTimer = null;

function setupBloomMusic() {

    if (!bloomMusic) {
        return;
    }

    if (
        !BLOOM_BIRTHDAY.music ||
        !BLOOM_BIRTHDAY.music.enabled
    ) {
        return;
    }

    bloomMusic.src =
        BLOOM_BIRTHDAY.music.source;

    bloomMusic.preload = "auto";

    bloomMusic.volume = 0;

    bloomMusic.loop = false;
}


function fadeMusicIn() {

    if (!bloomMusic) {
        return;
    }

    const duration =
        (BLOOM_BIRTHDAY.music.fadeIn || 2.5) * 1000;

    clearInterval(musicFadeTimer);

    bloomMusic.volume = 0;

    const steps = 50;
    const stepTime = duration / steps;
    let step = 0;

    musicFadeTimer = setInterval(() => {

        step++;

        bloomMusic.volume =
            Math.min(step / steps, 1);

        if (step >= steps) {

            clearInterval(musicFadeTimer);
            musicFadeTimer = null;

        }

    }, stepTime);
}


function startBloomMusic() {

    if (!bloomMusic) {
        return;
    }

    if (
        !BLOOM_BIRTHDAY.music ||
        !BLOOM_BIRTHDAY.music.enabled
    ) {
        return;
    }

    clearInterval(musicFadeTimer);

    bloomMusic.pause();

    bloomMusic.currentTime =
        BLOOM_BIRTHDAY.music.startAt || 0;

    bloomMusic.volume = 0;

    const playback =
        bloomMusic.play();

    if (playback) {

        playback
            .then(() => {

                fadeMusicIn();

            })
            .catch(error => {

                console.warn(
                    "BLOOM music could not start:",
                    error
                );

            });

    }
}


function fadeMusicOut() {

    if (!bloomMusic) {
        return;
    }

    const duration =
        (BLOOM_BIRTHDAY.music.fadeOut || 4) * 1000;

    clearInterval(musicFadeTimer);

    const startingVolume =
        bloomMusic.volume;

    const steps = 50;
    const stepTime = duration / steps;
    let step = 0;

    musicFadeTimer = setInterval(() => {

        step++;

        bloomMusic.volume =
            Math.max(
                startingVolume * (1 - step / steps),
                0
            );

        if (step >= steps) {

            clearInterval(musicFadeTimer);
            musicFadeTimer = null;

            bloomMusic.pause();
            bloomMusic.currentTime = 0;

        }

    }, stepTime);
}


setupBloomMusic();


/* =========================================================
   BLOOM AUTOMATION CLOCK
   ========================================================= */

let bloomAutomationStarted = false;

const BLOOM_TIMELINE = {

    note: 3,

    cake: 10,

    memories: 15,

    heart: 27,

    ending: 36

};


const bloomAutomationState = {

    note: false,

    cake: false,

    memories: false,

    heart: false,

    ending: false

};


function startBloomAutomation() {

    if (bloomAutomationStarted) {
        return;
    }

    bloomAutomationStarted = true;

    requestAnimationFrame(
        updateBloomAutomation
    );

}


function updateBloomAutomation() {

    if (
        !bloomMusic ||
        bloomMusic.paused
    ) {

        requestAnimationFrame(
            updateBloomAutomation
        );

        return;

    }


    const songTime =
        bloomMusic.currentTime -
        (BLOOM_BIRTHDAY.music.startAt || 0);


    /* -----------------------------------------------------
       NOTE / ENVELOPE
       ----------------------------------------------------- */

    if (
        songTime >= BLOOM_TIMELINE.note &&
        !bloomAutomationState.note
    ) {

        bloomAutomationState.note = true;

        openEnvelope();

    }


    /* -----------------------------------------------------
       CAKE
       ----------------------------------------------------- */

    if (
        songTime >= BLOOM_TIMELINE.cake &&
        !bloomAutomationState.cake
    ) {

        bloomAutomationState.cake = true;

        transitionTo(
            STATES.CAKE
        );

    }


    /* -----------------------------------------------------
       MEMORIES
       ----------------------------------------------------- */

    if (
        songTime >= BLOOM_TIMELINE.memories &&
        !bloomAutomationState.memories
    ) {

        const memories =
            getAvailableMemories();

        if (memories.length) {

            bloomAutomationState.memories = true;

            startMemorySequence(
                memories
            );

        }

    }


    /* -----------------------------------------------------
       HEART
       ----------------------------------------------------- */

    if (
        songTime >= BLOOM_TIMELINE.heart &&
        !bloomAutomationState.heart
    ) {

        const started =
            beginMemoryHeartMorph();

        if (started !== false) {

            bloomAutomationState.heart = true;

        }

    }


    /* -----------------------------------------------------
       ENDING
       ----------------------------------------------------- */

    if (
        songTime >= BLOOM_TIMELINE.ending &&
        !bloomAutomationState.ending
    ) {

        bloomAutomationState.ending = true;

        beginBloomEnding();

    }


    requestAnimationFrame(
        updateBloomAutomation
    );

}

let mascotMessageTimer = null;


function showMascotMessage(
    message,
    duration = 3200
) {

    if (!mascotBubble) {
        return;
    }

    clearTimeout(
        mascotMessageTimer
    );

    mascotBubble.textContent =
        message;

    mascotBubble.classList.remove(
        "is-visible"
    );

    requestAnimationFrame(() => {

        mascotBubble.classList.add(
            "is-visible"
        );

    });

    if (duration > 0) {

        mascotMessageTimer =
            setTimeout(() => {

                hideMascotMessage();

            }, duration);

    }

}


function hideMascotMessage() {

    if (!mascotBubble) {
        return;
    }

    clearTimeout(
        mascotMessageTimer
    );

    mascotBubble.classList.remove(
        "is-visible"
    );

}

/* =========================================================
   ORIGINAL BLOOM PETALS
   ========================================================= */

function createPetals(
    count = 12,
    transition = false
) {

    const petals =
        document.createElement("div");


    petals.className =
        transition
            ? "petals petals--transition"
            : "petals";


    for (
        let i = 0;
        i < count;
        i++
    ) {

        const petal =
            document.createElement("span");


        petal.className =
            "petal";


        petal.style.setProperty(
            "--petal-x",
            `${Math.random() * 100}%`
        );


        petal.style.setProperty(
            "--petal-size",
            `${7 + Math.random() * 10}px`
        );


        petal.style.setProperty(
            "--petal-duration",
            `${6 + Math.random() * 7}s`
        );


        petal.style.setProperty(
            "--petal-delay",
            `${Math.random() * -10}s`
        );


        petal.style.setProperty(
            "--petal-drift",
            `${-35 + Math.random() * 70}px`
        );


        petal.style.setProperty(
            "--petal-rotate",
            `${Math.random() * 360}deg`
        );


        petals.appendChild(
            petal
        );

    }


    return petals;

}


/* =========================================================
   ACTION BUTTON
   ========================================================= */

function setAction(
    label,
    callback
) {

    action.textContent =
        label;

    action.onclick =
        callback;


    /*
     * Once BLOOM automation has started,
     * scene actions are controlled by the
     * music clock instead of manual clicks.
     */

    if (
        bloomAutomationStarted &&
        currentState !== STATES.INTRO
    ) {

        action.hidden =
            true;

        action.disabled =
            true;

        return;

    }


    action.hidden =
        false;

    action.disabled =
        false;

}


/* =========================================================
   STOP MEMORY SYSTEM
   ========================================================= */

function stopMemorySystem() {

    if (
        cakePetalInterval
    ) {

        clearInterval(
            cakePetalInterval
        );

        cakePetalInterval =
            null;

    }


    memoryTimers.forEach(
        timer => clearTimeout(timer)
    );


    memoryTimers = [];


    memoryCards = [];

    memoryOpenedCount = 0;

}

                        class="note__content"

/* =========================================================
   OPEN ENVELOPE
   ========================================================= */

function openEnvelope() {

    if (
        envelopeOpened
    ) {

        return;

    }


    envelopeOpened =
        true;


    action.disabled =
        true;

    action.hidden =
        true;


    scene.classList.add(
        "is-envelope-opening"
    );


    setTimeout(
        () => {

            scene.classList.add(
                "is-postcard-visible"
            );

        },
        550
    );


    setTimeout(
        () => {

            scene.classList.add(
                "is-writing-visible"
            );


            startNoteWriting();

        },
        1450
    );

}


/* =========================================================
   NOTE WRITING
   ========================================================= */

function startNoteWriting() {

    showMascotMessage(
        "Shall we continue?",
        0
    );


    const lines =
        [
            ...document.querySelectorAll(
                ".note__writing .writing-line"
            )
        ];


    if (
        !lines.length
    ) {

        finishNoteWriting();

        return;

    }


    lines.forEach(
        (
            line,
            index
        ) => {

            setTimeout(
                () => {

                    line.classList.add(
                        "is-writing"
                    );

                },
                index * 250
            );

        }
    );


    const lastLine =
        lines.length - 1;


    const totalTime =
        (lastLine * 250) + 900;


    setTimeout(
        finishNoteWriting,
        totalTime
    );

}


/* =========================================================
   NOTE FINISHED
   ========================================================= */

function finishNoteWriting() {

    hideMascotMessage();


    if (
        bloomAutomationStarted
    ) {

        action.hidden =
            true;

        action.disabled =
            true;

        return;

    }


    action.disabled =
        false;


    setAction(
        "Continue",
        () => {

            hideMascotMessage();

            transitionTo(
                STATES.CAKE
            );

        }
    );

}


/* =========================================================
   AVAILABLE MEMORIES
   ========================================================= */

function getAvailableMemories() {

    return (
        BLOOM_BIRTHDAY.photos || []
    )

        .map(
            (
                src,
                index
            ) => ({

                src,

                index,

                mystery:
                    index >= 5

            })
        )

        .filter(
            item =>
                typeof item.src === "string" &&
                item.src.trim() !== ""
        );

}

function startMemorySequence(
    memories
) {

    /*
     * Memory scene begins immediately.
     * The music automation clock decides
     * when this function is called.
     */

    scene.classList.add(
        "is-memory-focus"
    );


    /*
     * Memories appear one after another.
     * Keep the spacing between them.
     */

    memories.forEach(

/* =========================================================
   CLOSING FLAKE STORM
   ========================================================= */

function startClosingFlakeStorm() {

    return;


    scene.classList.add(
        "is-closing-storm"
    );


    const oldLayer =
        scene.querySelector(
            ".closing-flake-layer"
        );


    if (oldLayer) {
        oldLayer.remove();
    }


    const oldAccumulation =
        scene.querySelector(
            ".closing-accumulation"
        );


    if (oldAccumulation) {
        oldAccumulation.remove();
    }


    /* =====================================================
       FALLING FLAKES
       ===================================================== */

    const layer =
        document.createElement("div");


    layer.className =
        "closing-flake-layer";


    scene.appendChild(
        layer
    );


    function createClosingFlake() {

        const flake =
            document.createElement("div");


        const roll =
            Math.random();


        let type;


        if (roll < 0.62) {

            type = "pink";

        } else if (roll < 0.88) {

            type = "amber";

        } else {

            type = "blue";

        }


        flake.className =
            `closing-flake closing-flake--${type}`;


        const x =
            Math.random() * 100;


        const size =
            10 +
            Math.random() * 18;


        const duration =
            2.6 +
            Math.random() * 2.4;


        const delay =
            Math.random() * .8;


        flake.style.setProperty(
            "--close-x",
            `${x}vw`
        );


        flake.style.setProperty(
            "--close-size",
            `${size}px`
        );


        flake.style.setProperty(
            "--close-duration",
            `${duration}s`
        );


        flake.style.setProperty(
            "--close-delay",
            `${delay}s`
        );


        flake.style.setProperty(
            "--close-rotate",
            `${Math.random() * 360}deg`
        );


        flake.style.setProperty(
            "--close-drift",

        );

            `${.72 + Math.random() * .5}`

        );

        setInterval(

                for (

                    i++

                    createClosingFlake();

                    stormCount++;

                }

                    stormCount >= 330
                ) {

                    clearInterval(

                    );

                }

            },
            90
        );


    /* =====================================================

       ===================================================== */

    const accumulation =
        document.createElement("div");

    accumulation.innerHTML = `

        ></div>

            class="closing-accumulation__flakes"
        ></div>

    `;


    scene.appendChild(
        accumulation
    );


    const accumulationFlakes =
        accumulation.querySelector(
            ".closing-accumulation__flakes"
        );


    for (
        let i = 0;
        i < 180;
        i++
    ) {

        const flake =
            document.createElement("span");


        const roll =
            Math.random();


        let type;


        if (roll < .62) {

            type = "pink";

        } else if (roll < .88) {

            type = "amber";

        } else {

            type = "blue";

        }


        flake.className =
            `accumulation-flake accumulation-flake--${type}`;


        flake.style.setProperty(
            "--pile-x",
            `${Math.random() * 100}%`
        );


        flake.style.setProperty(
            "--pile-y",
            `${Math.random() * 100}%`
        );


        flake.style.setProperty(
            "--pile-size",
            `${7 + Math.random() * 18}px`
        );


        flake.style.setProperty(
            "--pile-rotation",
            `${Math.random() * 360}deg`
        );


        flake.style.setProperty(
            "--pile-depth",
            `${.55 + Math.random() * .45}`
        );


        accumulationFlakes.appendChild(
            flake
        );

    }


    requestAnimationFrame(() => {

        accumulation.classList.add(
            "is-rising"
        );

    });


    setTimeout(
        () => {

            scene.classList.add(
                "is-closing-buried"
            );

        },
        6000
    );

}


/* =========================================================
   SCENE TRANSITIONS
   ========================================================= */

function transitionTo(
    nextState
) {

    if (
        isTransitioning
    ) {

        return;

    }


    isTransitioning =
        true;


    action.disabled =
        true;


    /* -----------------------------------------------------
       INTRO → ENVELOPE
       ----------------------------------------------------- */

    if (
        currentState === STATES.INTRO &&
        nextState === STATES.ENVELOPE
    ) {

        scene.classList.add(
            "is-petal-transition"
        );


        setTimeout(
            () => {

                scene.classList.add(
                    "is-flower-dissolving"
                );

            },
            180
        );


        setTimeout(
            () => {

                renderEnvelope();


                scene.classList.add(
                    "is-envelope-entering"
                );


                requestAnimationFrame(
                    () => {

                        requestAnimationFrame(
                            () => {

                                scene.classList.remove(
                                    "is-envelope-entering"
                                );

                            }
                        );

                    }
                );

            },
            950
        );


        setTimeout(
            () => {

                isTransitioning =
                    false;

            },
            1800
        );


        return;

    }


    /* -----------------------------------------------------
       ENVELOPE → CAKE
       ----------------------------------------------------- */

    if (
        currentState === STATES.ENVELOPE &&
        nextState === STATES.CAKE
    ) {

        renderCake();


        isTransitioning =
            false;


        return;

    }

}
